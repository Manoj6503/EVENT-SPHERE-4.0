cmake_minimum_required(VERSION 3.14)
project(EventSphere CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
add_executable(eventsphere src/main.cpp)
target_include_directories(eventsphere PRIVATE include)
target_link_libraries(eventsphere SQLite::SQLite3 OpenSSL::Crypto Threads::Threads CURL::libcurl)

# Automated tests:  cmake -S backend -B build && cmake --build build && ctest --test-dir build --output-on-failure
option(ES_TESTS "Build and register the automated tests" ON)
if(ES_TESTS)
  enable_testing()
  add_executable(core_tests tests/test_core.cpp)
  target_include_directories(core_tests PRIVATE include src)
  target_link_libraries(core_tests SQLite::SQLite3 OpenSSL::Crypto Threads::Threads CURL::libcurl)
  add_test(NAME core_unit_tests COMMAND core_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    get_filename_component(ES_ROOT "${CMAKE_SOURCE_DIR}/.." ABSOLUTE)
    add_test(NAME api_integration_tests COMMAND ${Python3_EXECUTABLE} ${ES_ROOT}/tests/api_test.py --server $<TARGET_FILE:eventsphere> --root ${ES_ROOT})
    add_test(NAME openapi_matches_routes COMMAND ${Python3_EXECUTABLE} ${ES_ROOT}/tests/check_openapi.py --root ${ES_ROOT})
  endif()
endif()
